dijkstra.cpp
O((V+E) log V)
Dijkstra Shortest Path
Observation: Min-heap priority queue with {dist, u}. Non-negative edges. 
vector<ll> dijkstra(int s, int n, vector<vector<pair<int,ll>>>& g) {
  vector<ll> d(n, INF); d[s] = 0;
  priority_queue<pair<ll,int>, vector<pair<ll,int>>, greater<>> pq;
  pq.push({0, s});
  while (!pq.empty()) {
    auto [dist_u, u] = pq.top(); pq.pop();
    if (dist_u > d[u]) continue;
    for (auto& [v, w] : g[u])
      if (d[u] + w < d[v]) { d[v] = d[u] + w; pq.push({d[v], v}); }
  }
  return d;
}

01. DATA STRUCTURES & TREES
segtree.cpp
O(log N)
Iterative Segment Tree
Observation: Point update and range sum query. Zero-indexed leaves at tree[p + n]. 
template<class T> struct SegTree {
  int n; vector<T> tree;
  SegTree(int n) : n(n), tree(2 * n, 0) {}
  void update(int p, T val) {
    for (tree[p += n] = val; p > 1; p >>= 1)
      tree[p >> 1] = tree[p] + tree[p ^ 1];
  }
  T query(int l, int r) {
    T res = 0;
    for (l += n, r += n; l < r; l >>= 1, r >>= 1) {
      if (l & 1) res += tree[l++];
      if (r & 1) res += tree[--r];
    }
    return res;
  }
};
